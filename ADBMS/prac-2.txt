use libraryDB;
db.books.drop();

// Sample documents
db.books.insertMany([
  { title: "DSA Handbook", price: 650, copies: 12, category: "Computer" },
  { title: "Operating Systems", price: 550, copies: 8, category: "Computer" },
  { title: "Microeconomics", price: 400, copies: 20, category: "Economics" },
  { title: "DSA Handbook", price: 650, copies: 6, category: "Computer" },
  { title: "Indian Polity", price: 300, copies: 15, category: "Politics" }
]);

// ---------------- Aggregation Framework ----------------

// Total copies available for each book title
db.books.aggregate([
  { $group: { _id: "$title", totalCopies: { $sum: "$copies" } } }
]);

// Average price of books category wise
db.books.aggregate([
  { $group: { _id: "$category", avgPrice: { $avg: "$price" } } }
]);

// Total revenue possible per category and title
db.books.aggregate([
  {
    $group: {
      _id: { category: "$category", title: "$title" },
      totalRevenue: { $sum: { $multiply: ["$price", "$copies"] } }
    }
  }
]);

// ---------------- MapReduce ----------------

var mapFunction = function() {
  emit(this.category, this.copies);
};

var reduceFunction = function(key, values) {
  return Array.sum(values);
};

db.books.mapReduce(
  mapFunction,
  reduceFunction,
  { out: "totalCopiesMapReduce" }
);

db.totalCopiesMapReduce.find();

// ---------------- Indexing ----------------

// Create single field index
db.books.createIndex({ title: 1 });

// Create compound index
db.books.createIndex({ title: 1, category: 1 });

// Drop an index
db.books.dropIndex({ title: 1 });

// List all indexes
db.books.getIndexes();

// Explain query with index
db.books.find({ title: "DSA Handbook" }).explain("executionStats");
